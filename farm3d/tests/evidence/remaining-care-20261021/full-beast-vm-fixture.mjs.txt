// Actual Beast + accepted rigs and planners; read-only ownership replaces the DOM game shell.
// This is motion/routing evidence, not a renderer, native-input, save or economy proof.
import fs from 'node:fs';import vm from 'node:vm';import * as T from '../lib/three.module.min.js';import * as Live3D from '../live3d.js';import {farmGroundY}from'../farm-land.js';import {cowYardLayout}from'../cow-yard.js';import {CHICKEN_YARD_LAYOUT}from'../chicken-yard.js';import {createRemainingCareYards}from'../remaining-care-yard.js';import {remainingCareLayouts}from'../remaining-care-layout.js';import {createRemainingCareController}from'../remaining-care-controller.js';
export function remainingWorld(seed=4242){
 const html=fs.readFileSync(new URL('../index.html',import.meta.url),'utf8'),layouts=remainingCareLayouts(),yards=createRemainingCareYards(),f=JSON.parse(fs.readFileSync(new URL('./fixtures/remaining-care-initial-world.json',import.meta.url),'utf8')),blocks=[...f.existingBlocks,...Object.values(yards).flatMap(y=>y.obstacles)],runs=vm.runInNewContext(html.slice(html.indexOf('const FENCE_RUNS ='),html.indexOf('const fenceSegs ='))+';FENCE_RUNS'),fences=[];
 for(const r of runs)for(let i=1;i<r.length;i++)fences.push([...r[i-1],...r[i]]);fences.push(...CHICKEN_YARD_LAYOUT.fenceSegments,...Object.values(yards).flatMap(y=>y.fences));
 const animals=[],pens=Object.fromEntries(['horse','sheep'].map(k=>{const l=layouts[k],p=l.turnouts[0];return[k,{poly:l.polygon,care:k,kinds:[k],bed:[p[0],p[2],2],shelter:{x:[l.shelter.x-1,l.shelter.x+1],z:[l.shelter.z-1,l.shelter.z+1]},water:[l.stations[2].dock[0],l.stations[2].dock[2],.3,.3]}];})),PENS={courtyard:{poly:cowYardLayout(25).polygon},paddock:{poly:[[4.5,-8.4],[12.8,-8.4],[12.6,-1.2],[10.81,10.48],[4.5,8.87]]},yard:{poly:CHICKEN_YARD_LAYOUT.polygon}},S={pens:{horse:{list:Array(4).fill(0)},sheep:{list:Array(5).fill(0)}}},materials=new Map();
 const c={THREE:T,Live3D,G:{S},remainingSites:Object.fromEntries(Object.keys(layouts).map(k=>[k,{active:true}])),remainingLayouts:layouts,cowYardActive:false,PRESET:{},clockT:0,tapPoint:null,animals,scene:new T.Scene(),contact(){},groundY:farmGroundY,PENS,PEN_LIST:[PENS.paddock,PENS.yard,pens.horse,pens.sheep],penOf:k=>pens[k]||null,BLOCKS:blocks,HOME:new T.Vector2(1,3),ROAM:19,fenceSegs:fences,sensePlayer(){},emoteMat(e){if(!materials.has(e))materials.set(e,new T.SpriteMaterial());return materials.get(e);}};
 c.OBST=[...blocks.map(([x,z,hw,hd])=>({x,z,hw,hd,c:1,s:0})),...fences.map(([ax,az,bx,bz])=>{const a=Math.atan2(bz-az,bx-ax);return{x:(ax+bx)/2,z:(az+bz)/2,hw:Math.hypot(bx-ax,bz-az)/2,hd:.1,c:Math.cos(a),s:Math.sin(a)};})];
 vm.createContext(c);vm.runInContext(html.match(/function rng\(seed\) \{[^\n]+/)[0]+';globalThis.rnd=rng('+seed+');',c);
 vm.runInContext(html.slice(html.indexOf('const PERSONALITY ='),html.indexOf('// where each kind lives:')),c);
 vm.runInContext(html.slice(html.indexOf('function edgeDepths('),html.indexOf('// where the animals may not walk:')),c);
 vm.runInContext(html.slice(html.indexOf('const toLocal ='),html.indexOf('// little mood bubbles')),c);
 c.acceptsPoint=vm.runInContext('fitsPen',c);
 vm.runInContext('globalThis.Beast='+html.slice(html.indexOf('class Beast {'),html.indexOf('let clockT =')),c);
 c.remainingCare=createRemainingCareController({layouts,neighbours:()=>animals,active:()=>true,obstacles:()=>blocks,segments:()=>fences,groundY:farmGroundY,forbiddenPolygons:a=>a.pen?[]:c.PEN_LIST.map(p=>p.poly),clearPoint:(a,x,z)=>c.acceptsPoint(a.pen,x,z,layouts[a.kind].margin),updateRig:(r,dx,dz,dt)=>Live3D.updateLiveAnimal(r,dx,dz,dt,'moving',false)});
 for(const kind of ['sheep','horse','dog','cat'])for(let i=0;i<layouts[kind].capacity;i++){const p=layouts[kind].turnouts[i],pet=kind==='dog'||kind==='cat',name=pet?(kind==='dog'?'Buddy':'Whiskers'):kind+' Care '+i,ref=pet?{type:'pet',kind,j:kind==='dog'?0:1}:{type:'pen',kind,i},a=new c.Beast(kind,name,vm.runInContext('persOf('+JSON.stringify(name)+')',c),p[0],p[2],ref);a.need={hunger:0,tired:0,lonely:0};a.remainingDrinkAt=Infinity;a.remainingMealAt=Infinity;animals.push(a);}
 return{c,layouts,animals,step(dt=1/30){c.clockT+=dt;for(const a of animals){const before=a.g.position.clone(),care=!!a.remainingTask;a.update(dt,c.clockT);if((care||a.remainingTask)&&a.g.position.distanceTo(before)>layouts[a.kind].speed*dt+1e-7)throw Error('Care speed exceeded');if(Math.abs(a.g.position.y-farmGroundY(a.g.position.x,a.g.position.z))>1e-7)throw Error('Unsupported root '+a.kind);}},report(){return animals.map(a=>({kind:a.kind,i:a.ref.i,j:a.ref.j,p:a.g.position.toArray(),stage:a.remainingTask?.stage,station:a.remainingTask?.station.id,path:a.remainingTask?.path,blocked:a.remainingTask?.blocked,act:a.act,label:a.label}));},dispose(){animals.forEach(a=>a.rig3d.dispose());Object.values(yards).forEach(y=>y.dispose());materials.forEach(m=>m.dispose());}};
}
