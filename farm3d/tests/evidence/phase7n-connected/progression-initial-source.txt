// Accelerated consecutive progression in the real local game. No injected
// levels, XP, currency or inventories; only the clock and random seed are fixed.
const fs=require('node:fs'),path=require('node:path'),assert=require('node:assert/strict'),crypto=require('node:crypto');
const {start}=require('./qualification-cdp.cjs');
const ref=process.env.QUALIFICATION_REF||'e2f2ab2edfdabdd0d9d78b896c21cad50fb82a9c';
const out=path.resolve(process.env.TEST_ARTIFACTS||path.join(__dirname,'artifacts','qualification-progression'));
let h,stage='launch';const results=[];
(async()=>{let failed=true;try{
  h=await start(out,ref);
  await h.cdp.send('Page.addScriptToEvaluateOnNewDocument',{source:`if(!sessionStorage.getItem('__qualifiedSeed')){localStorage.setItem('sa3d-guest','1');localStorage.setItem('sa3d-seen','1');sessionStorage.setItem('__qualifiedSeed','1');}window.__qualificationTime=Date.now();Date.now=()=>window.__qualificationTime;let seed=1729;Math.random=()=>{seed=(Math.imul(seed,1664525)+1013904223)>>>0;return seed/4294967296;};`});
  await h.boot();stage='consecutive production/order/unlock progression';
  const progression=await h.cdp.evaluate(`(()=>{
    const G=__dbg.G,S=G.S,log=[],recipes=new Set(),levels=new Set([1]);let operations=0;
    const check=(v,m)=>{if(!v)throw Error(m);};
    const button=selector=>{const el=document.querySelector(selector);check(el&&!el.disabled,'Missing/enabled control: '+selector);el.click();};
    const bounded=()=>check(++operations<10000,'bounded operation count');
    G.close();G.opts.quiet=true;S.tut=99;S.nextEventAt=9e15;S.nextRushAt=9e15;S.nextVisitorAt=9e15;
    check(S.level===1&&S.coins===60&&S.xp===0,'fresh starting progress');
    check(!G.wear('hairColor',6),'level 8 hair colour locked at level 1');
    const best=()=>Object.keys(G.CROPS).filter(id=>G.ITEMS[id].lvl<=S.level).sort((a,b)=>G.CROPS[b].xp-G.CROPS[a].xp)[0];
    function sell(id){if(!G.have(id))return;G.openPanel('barn');button('[data-act="selItem"][data-id="'+id+'"]');const controls=[...document.querySelectorAll('[data-act="sell"][data-id="'+id+'"]')];controls.sort((a,b)=>+b.dataset.n-+a.dataset.n);check(controls.length,'sell controls');controls[0].click();G.close();}
    function room(){if(G.space()<4){const id=Object.keys(S.barn).sort((a,b)=>G.have(b)-G.have(a))[0];check(id,'space recovery item');sell(id);}check(G.space()>=2,'enough harvest space');}
    function reward(xp,coins=0,gems=0){let x=S.xp+xp,l=S.level,c=S.coins+coins,g=S.gems+gems;while(x>=G.xpNeed(l)){x-=G.xpNeed(l);l++;c+=20*l;g+=2;}return {xp:x,level:l,coins:c,gems:g};}
    function verify(r){for(const k of ['xp','level','coins','gems'])check(S[k]===r[k],'exact '+k+' reward '+S[k]+' != '+r[k]);levels.add(S.level);}
    function grow(id){bounded();room();const old=G.have(id);check(G.plant(0,id,true),'plant '+id);const r=reward(G.CROPS[id].xp);window.__qualificationTime=S.plots[0].end+1;check(G.harvest(0,true)===true,'harvest '+id);verify(r);check(G.have(id)>=old+1,'net crop reward '+id);G.save();}
    function money(cost){for(let i=0;S.coins<cost&&i<1000;i++){const id=best();grow(id);sell(id);}check(S.coins>=cost,'finance '+cost);}
    function building(bid){if(S.buildings[bid].owned)return;const b=G.BUILDINGS[bid];check(S.level>=b.lvl,'building unlock '+bid);money(b.cost);G.openPanel('lot',bid);button('[data-act="building"][data-k="'+bid+'"]');check(S.buildings[bid].owned,'building purchased '+bid);G.close();}
    function pen(kind){if(S.pens[kind].owned)return;const a=G.ANIMALS[kind];check(S.level>=a.lvl,'pen unlock '+kind);money(a.penCost);G.openPanel('penLot',kind);button('[data-act="pen"][data-k="'+kind+'"]');check(S.pens[kind].owned,'pen purchased '+kind);G.close();}
    function ensure(id,n){bounded();check(G.ITEMS[id].lvl<=S.level,'item unlock '+id);while(G.have(id)<n){
      if(G.CROPS[id]){grow(id);continue;}
      const kind=Object.keys(G.ANIMALS).find(k=>G.ANIMALS[k].out===id);
      if(kind){pen(kind);ensure(G.ANIMALS[kind].feed,1);check(G.feedOne(kind,0),'animal fed '+kind);window.__qualificationTime=S.pens[kind].list[0]+1;check(G.collectAnimal(kind,0),'animal collected '+kind);continue;}
      const bid=Object.keys(G.BUILDINGS).find(k=>G.BUILDINGS[k].recipes.includes(id)),r=G.RECIPES[id];building(bid);
      for(const [input,qty]of Object.entries(r.in))ensure(input,qty);
      const inputs=Object.fromEntries(Object.keys(r.in).map(k=>[k,G.have(k)]));G.openPanel('building',bid);button('[data-act="make"][data-b="'+bid+'"][data-r="'+id+'"]');
      for(const [input,qty]of Object.entries(r.in))check(G.have(input)===inputs[input]-qty,'consumed '+input+' for '+id);
      const j=S.buildings[bid].jobs[0];check(j&&j.r===id,'queued '+id);window.__qualificationTime=j.end+1;G.renderPanel();const count=G.have(id);button('[data-act="collectJobs"][data-b="'+bid+'"]');check(G.have(id)===count+r.out,'output '+id);recipes.add(id);G.close();
    }}
    function order(){const i=S.orders.findIndex(o=>!o.wait),o=structuredClone(S.orders[i]);check(i>=0,'generated order');for(const [id,n]of Object.entries(o.items))ensure(id,n);
      const inventory=structuredClone(S.barn),orderCount=S.stats.orders,expected=reward(o.xp,o.coins,o.gem+((orderCount+1)%10===0?1:0));G.openPanel('orders');button('[data-act="deliver"][data-i="'+i+'"]');verify(expected);check(S.stats.orders===orderCount+1,'one order counted');for(const [id,n]of Object.entries(o.items))check(G.have(id)===(inventory[id]||0)-n,'order consumption '+id);check(S.orders[i].wait>G.now(),'order cooldown');G.close();log.push({kind:'generated-order',items:o.items,level:S.level});}
    for(let target=1;target<=8;target++){
      while(S.level<target){grow(best());if(G.have(best())>12)sell(best());}
      check(S.level===target,'consecutive level '+target);levels.add(target);
      for(const [id,r]of Object.entries(G.RECIPES))if(G.ITEMS[id].lvl===target){ensure(id,r.out);log.push({kind:'first-recipe',id,level:S.level});sell(id);}
      order();
      if(target===5){check(S.perks.owed===1,'level 5 perk awarded');G.openPanel('perk');button('[data-act="perk"][data-k="grow"]');check(S.perks.grow===1&&S.perks.owed===0,'perk picked exactly once');}
      if(target===2)check(G.wear('shirt',3),'level 2 wardrobe unlock');
      if(target===5)check(G.wear('shirt',4),'level 5 wardrobe unlock');
      if(target===8)check(G.wear('hairColor',6),'level 8 wardrobe unlock');
      G.save();log.push({kind:'level-checkpoint',target,level:S.level,xp:S.xp,coins:S.coins,gems:S.gems,harvests:S.stats.harvests,orders:S.stats.orders,made:S.stats.made});
    }
    check(recipes.size===Object.keys(G.RECIPES).length,'all 18 recipes actually produced');check(S.stats.orders===8,'one generated order per target level');
    G.close();G.save();return {levels:[...levels],recipes:[...recipes],operations,log,raw:localStorage.getItem(G.SAVE_KEY),look:G.lookOf(),stats:structuredClone(S.stats)};
  })()`);
  assert.deepEqual(progression.levels,[1,2,3,4,5,6,7,8]);assert.equal(progression.recipes.length,18);results.push({name:stage,status:'PASS',detail:progression});console.log('PASS consecutive levels 1-8, 18 recipes and eight generated orders');
  stage='progression reload';await h.reload();const after=await h.cdp.evaluate(`(()=>{const s=JSON.parse(localStorage.getItem(__dbg.G.SAVE_KEY));delete s.lastSeen;delete s.lastNag;return s;})()`);const before=JSON.parse(progression.raw);delete before.lastSeen;delete before.lastNag;assert.deepEqual(after,before);assert.deepEqual(h.errors,[]);results.push({name:stage,status:'PASS',detail:{farmExactExceptLifecycleTimestamps:true}});await h.cdp.screenshot(path.join(out,'level-eight.png'));failed=false;
}catch(e){results.push({name:stage,status:'FAIL',error:e.stack});console.error(e.stack);process.exitCode=1;if(h)try{await h.cdp.screenshot(path.join(out,'first-failure.png'));}catch{}}
finally{if(h)await h.close();fs.mkdirSync(out,{recursive:true});fs.writeFileSync(path.join(out,'results.json'),JSON.stringify({status:failed?'FAIL':'PASS',sourceRef:ref,helperSha256:crypto.createHash('sha256').update(fs.readFileSync(__filename)).digest('hex'),node:process.version,limits:'Accelerated clock, seeded randomness, popup suppression, static scene. Game APIs handle crops/animals; rendered DOM buttons handle production, orders, shop and perks. No wall-clock/balance, movement, long-session or device performance claim.',results},null,2));}})();
